"""Drives CPR Coach on the phone through adb. Never touches the phone app beyond confirming the number shown."""
import re
import subprocess
import sys
import time
import xml.etree.ElementTree as ET
from pathlib import Path

ADB = str(Path.home() / "AppData/Local/Android/Sdk/platform-tools/adb.exe")
SERIAL = "EILZRCFQLZIZ6H6P"
PACKAGE = "com.zeroflux.cprcoach"
OUT = Path(__file__).parent / "phone"
OUT.mkdir(exist_ok=True)


def adb(*args, check=False):
    return subprocess.run([ADB, "-s", SERIAL, *args], capture_output=True, text=True, encoding="utf-8",
                          errors="replace", timeout=60)


def shot(name):
    with open(OUT / f"{name}.png", "wb") as f:
        subprocess.run([ADB, "-s", SERIAL, "exec-out", "screencap", "-p"], stdout=f, timeout=60)


def nodes():
    adb("shell", "uiautomator", "dump", "/sdcard/ui.xml")
    xml = adb("shell", "cat", "/sdcard/ui.xml").stdout
    try:
        root = ET.fromstring(xml[xml.index("<"):])
    except (ET.ParseError, ValueError):
        return []
    found = []
    for n in root.iter("node"):
        m = re.match(r"\[(\d+),(\d+)\]\[(\d+),(\d+)\]", n.get("bounds", ""))
        if m:
            x1, y1, x2, y2 = map(int, m.groups())
            found.append({"text": n.get("text", ""), "desc": n.get("content-desc", ""), "pkg": n.get("package", ""),
                          "click": n.get("clickable") == "true", "box": (x1, y1, x2, y2)})
    return found


def texts():
    return [n["text"] or n["desc"] for n in nodes() if n["text"] or n["desc"]]


def top_package():
    out = adb("shell", "dumpsys", "activity", "activities").stdout
    m = re.search(r"topResumedActivity=\S+ \S+ (\S+?)/", out)
    return m.group(1) if m else "?"


def tap_text(label, long_ms=None):
    for n in nodes():
        if label == n["text"] or label == n["desc"]:
            x1, y1, x2, y2 = n["box"]
            x, y = (x1 + x2) // 2, (y1 + y2) // 2
            if long_ms:
                adb("shell", "input", "swipe", str(x), str(y), str(x), str(y), str(long_ms))
            else:
                adb("shell", "input", "tap", str(x), str(y))
            time.sleep(1.2)
            return True
    print("  NOT FOUND:", label)
    return False


def alive():
    return bool(adb("shell", "pidof", PACKAGE).stdout.strip())


def crashes():
    out = adb("logcat", "-d", "-b", "crash").stdout
    return [l for l in out.splitlines() if re.search(r"FATAL|Caused by|Fatal signal|SIGSEGV|SIGABRT", l)]


def step(name):
    time.sleep(1.2)
    shot(name)
    print(f"[{name}] alive={alive()} top={top_package()} :: {' | '.join(texts())[:300]}")


def main():
    adb("logcat", "-c")
    adb("shell", "am", "force-stop", PACKAGE)
    adb("shell", "input", "keyevent", "KEYCODE_WAKEUP")
    adb("shell", "am", "start", "-n", f"{PACKAGE}/.MainActivity")
    time.sleep(3)
    step("01_home_en")

    # Languages: chip opens the picker.
    tap_text("English")
    step("02_picker")
    tap_text("हिन्दी")
    step("03_home_hi")
    tap_text("हिन्दी")
    tap_text("ಕನ್ನಡ")
    step("04_home_kn")
    tap_text("ಕನ್ನಡ")
    tap_text("English")
    step("05_home_en_again")

    # Emergency, then the dialer. Never press call.
    tap_text("Emergency")
    step("06_emergency")
    tap_text("Call 112")
    time.sleep(1.5)
    shot("07_dialer")
    package = top_package()
    screen = " ".join(texts())
    print(f"[07_dialer] top={package} shows 112: {'112' in screen.replace(' ', '')} :: {screen[:200]}")
    adb("shell", "input", "keyevent", "KEYCODE_BACK")
    time.sleep(1.5)
    if top_package() != PACKAGE:
        adb("shell", "input", "keyevent", "KEYCODE_BACK")
        time.sleep(1.5)
    step("08_back_from_dialer")
    tap_text("Start compressions")
    step("09_compressing")
    time.sleep(8)
    step("10_compressing_later")
    print("metronome service:", "MetronomeService" in adb("shell", "dumpsys", "activity", "services", PACKAGE).stdout)
    tap_text("Hold to stop", long_ms=1400)
    step("11_after_stop")
    print("metronome service after stop:", "MetronomeService" in adb("shell", "dumpsys", "activity", "services", PACKAGE).stdout)

    # Practise.
    tap_text("Practise")
    step("12_practise_first")
    print("crashes so far:", crashes())


if __name__ == "__main__":
    main()
